```
PERMISSION CHECK ALGORITHM
???????????????????????????

function CanUserAccessComponent(userId, idComponent):
    userPermissions = GetUserPermissions(userId)
    componentPermissions = GetComponentPermissions(idComponent)
    
    if componentPermissions is empty:
        return true  // Public component
    
    intersection = userPermissions ? componentPermissions
    return intersection is not empty


function CanUserPerformAction(userId, actionKey):
    userPermissions = GetUserPermissions(userId)
    actionPermissions = GetActionPermissions(actionKey)
    
    if actionPermissions is empty:
        return true  // Public action
    
    if user.IsInRole("Admin"):
        return true  // Admin bypass
    
    intersection = userPermissions ? actionPermissions
    return intersection is not empty
```

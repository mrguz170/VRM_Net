use vrm

DELIMITER $$
CREATE PROCEDURE sp_get_all_user(
)
BEGIN
    SELECT 
		user_id as userId,
		user_name as UserName,
		CONCAT(name,' ',last_name,' ',second_last_name) as NombreCompleto, 
		email as Email, 
		b.role_name as role,
		a.role_id as RoleId ,
		a.is_active as IsActive
	FROM cat_users a inner join cat_roles b on a.role_id = b.role_id;
	
END $$
DELIMITER ;

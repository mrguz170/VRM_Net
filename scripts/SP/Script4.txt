CREATE PROCEDURE sp_get_user(
    IN login VARCHAR(30) 
)
BEGIN
    IF  login LIKE '%@%' THEN
		SELECT 
        user_id as userId,
		user_name as UserName,
		CONCAT(name,' ',last_name,' ',second_last_name) as NombreCompleto, 
		email , 
		b.role_name as role,
		a.role_id as RoleId , 
		password
			FROM cat_users a inner join cat_roles b on a.role_id = b.role_id
			WHERE a.Email = login; 
    ELSE
		SELECT 
        user_id,
        user_name,
		CONCAT(name,' ',last_name,' ',second_last_name) as NombreCompleto, 
		email ,
		b.role_name as role,
		a.role_id as RoleId  , 
		password
			FROM cat_users a inner join cat_roles b on a.role_id = b.role_id
			WHERE a.user_name = login;
    END IF;
END $$
